cmake_minimum_required(VERSION 3.22)
project(NoctomorphAudit VERSION 0.0.4 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
option(AUDIT_SANITIZERS "Address and undefined-behaviour sanitizers" OFF)
set(JUCE_SOURCE "" CACHE PATH "JUCE 9.0.4 checkout")
set(V4JB_ASSETS "" CACHE PATH "V4JB 2b3bc12400ad3337ba6d8f2e37dc61672c39b951 checkout")
set(PRODUCT_SOURCE "${CMAKE_CURRENT_LIST_DIR}/.." CACHE PATH "Product source under test")
add_subdirectory("${JUCE_SOURCE}" juce EXCLUDE_FROM_ALL)
file(GLOB factory_xml "${V4JB_ASSETS}/presets/*.xml")
file(GLOB factory_wav "${V4JB_ASSETS}/presets/waves/*.wav")
list(LENGTH factory_xml preset_count)
if(NOT preset_count EQUAL 494)
  message(FATAL_ERROR "Expected exactly 494 pinned factory presets")
endif()
juce_add_binary_data(AuditResources SOURCES ${factory_xml} ${factory_wav} "${PRODUCT_SOURCE}/assets/logo.svg")
juce_add_console_app(processor_check PRODUCT_NAME "Noctomorph audit")
juce_generate_juce_header(processor_check)
target_sources(processor_check PRIVATE processor_check.cpp "${PRODUCT_SOURCE}/src/PluginProcessor.cpp" "${PRODUCT_SOURCE}/src/PluginEditor.cpp")
target_include_directories(processor_check PRIVATE "${PRODUCT_SOURCE}/src")
target_compile_definitions(processor_check PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1 JucePlugin_VersionString="0.0.4" NOCTOMORPH_AUDIT_BUILD=1)
target_link_libraries(processor_check PRIVATE AuditResources juce::juce_audio_utils juce::juce_dsp)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set_source_files_properties(processor_check.cpp PROPERTIES COMPILE_DEFINITIONS AUDIT_WRAP_ALLOCATIONS=1)
  target_link_options(processor_check PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc)
endif()
foreach(name engine_check version_check)
  add_executable(${name} "${PRODUCT_SOURCE}/tests/${name}.cpp")
  target_include_directories(${name} PRIVATE "${PRODUCT_SOURCE}/src")
  add_test(NAME ${name} COMMAND ${name})
endforeach()
if(AUDIT_SANITIZERS AND NOT MSVC)
  foreach(target processor_check engine_check version_check)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined,float-cast-overflow -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined,float-cast-overflow)
  endforeach()
endif()
foreach(probe blocks keys wet tempo state gui waves realtime presets save program panic midi)
  add_test(NAME processor_${probe} COMMAND processor_check ${probe})
endforeach()
foreach(name transport_safety_check dsp_safety_check audio_read_slot_check)
  add_executable(${name} "${CMAKE_CURRENT_LIST_DIR}/${name}.cpp")
  target_include_directories(${name} PRIVATE "${PRODUCT_SOURCE}/src")
  if(AUDIT_SANITIZERS AND NOT MSVC)
    target_compile_options(${name} PRIVATE -fsanitize=address,undefined,float-cast-overflow -fno-omit-frame-pointer)
    target_link_options(${name} PRIVATE -fsanitize=address,undefined,float-cast-overflow)
  endif()
endforeach()
find_package(Threads REQUIRED)
target_link_libraries(audio_read_slot_check PRIVATE Threads::Threads)
add_test(NAME transport_safety_check COMMAND transport_safety_check)
add_test(NAME audio_read_slot_check COMMAND audio_read_slot_check)
foreach(probe version preroll ppq voices)
  add_test(NAME dsp_${probe} COMMAND dsp_safety_check ${probe})
endforeach()
